\subsubsection{Section A -- 1 mark each}
\begin{enumerate}[label=\textbf{Q\arabic*.},ref=\arabic*]

% Q1: the discriminant handles equal roots directly.  The nonzero leading
% coefficient check prevents the spurious p=0 result from being accepted.
\item If the quadratic equation $px^2-2\sqrt5\,px+15=0$ has two equal roots, find $p$. \hfill (CBSE 2015, 30/1, Q1)

\solution By \eqref{eq:quad-general}, the discriminant must be zero:
\begin{align}
(-2\sqrt5p)^2-4(p)(15)&=0\\
20p^2-60p&=0\\
20p(p-3)&=0.
\end{align}
The algebra gives $p=0$ or $p=3$. But $p=0$ removes the $x^2$ and $x$ terms and leaves $15=0$, which is impossible. Therefore
\begin{equation}
\boxed{p=3}.
\end{equation}

% Q2: tangent ratios connect the measured height and shadow to the angle.
\item In Figure~\ref{fig:class10-tower}, tower $AB$ is $20\,\mathrm m$ high and shadow $BC$ is $20\sqrt3\,\mathrm m$ long. Find the Sun's altitude. \hfill (CBSE 2015, 30/1, Q2)
\begin{figure}[H]\centering
  \includegraphics[width=0.83\linewidth]{figs/Tower}
  \caption{Official dimensions: height $20\,\mathrm m$, shadow $20\sqrt3\,\mathrm m$.}
  \label{fig:class10-tower}
\end{figure}
\solution Let the angle of elevation be $\theta$. The tower is perpendicular to the ground, so $ABC$ is right-angled at $B$. Using \eqref{eq:trig-ratios},
\begin{align}
\tan\theta&=\frac{AB}{BC}=\frac{20}{20\sqrt3}=\frac1{\sqrt3}.
\end{align}
Hence $\boxed{\theta=30^\circ}$. The radical in the shadow length is part of the official question; it is not $203\,\mathrm m$.

% Q3: different dice are distinguishable, so ordered outcomes are counted.
\item Two different dice are tossed together. Find the probability that the product of their top numbers is $6$. \hfill (CBSE 2015, 30/1, Q3)

\solution By \eqref{eq:prob-classical}, there are $36$ equally likely ordered outcomes. The product is $6$ only for
\[
(1,6),\ (6,1),\ (2,3),\ (3,2).
\]
There are four such outcomes; therefore $P=4/36=\boxed{1/9}$.

% Q4: use the radius-tangent right angle, then the central/inscribed-angle relation.
\item In Figure~\ref{fig:class10-chord-tangent}, $PQ$ is a chord of a circle with centre $O$ and $PT$ is a tangent. If $\angle QPT=60^\circ$, find $\angle PRQ$. \hfill (CBSE 2015, 30/1, Q4)
\begin{figure}[H]\centering
  \includegraphics[width=0.69\linewidth]{figs/TangentChord}
  \caption{Chord--tangent configuration as in the official paper.}
  \label{fig:class10-chord-tangent}
\end{figure}
\solution Since $OP\perp PT$, $\angle OPQ=90^\circ-60^\circ=30^\circ$. As $OP=OQ$, triangle $OPQ$ is isosceles, so $\angle OQP=30^\circ$ and $\angle POQ=120^\circ$. Point $R$ lies on the minor arc $PQ$, so $\angle PRQ$ subtends the major arc. Its measure is $360^\circ-120^\circ=240^\circ$. The angle at the circumference is half its intercepted arc, giving
\[
\boxed{\angle PRQ=120^\circ}.
\]
\end{enumerate}

\subsubsection{Section B -- 2 marks each}
\begin{enumerate}[label=\textbf{Q\arabic*.},ref=\arabic*,start=5]

% Q5: congruent right triangles establish equal tangents; the angle bisector
% then supplies a 30-60-90 triangle and the claimed length identity.
\item Two tangents $RQ$ and $RP$ are drawn from external point $R$ to a circle with centre $O$. If $\angle PRQ=120^\circ$, prove $OR=PR+RQ$. \hfill (CBSE 2015, 30/1, Q5)
\begin{figure}[H]\centering
  \includegraphics[width=0.65\linewidth]{figs/TwoTangents}
  \caption{Two tangents from an external point.}
  \label{fig:class10-two-tangents}
\end{figure}
\solution Radii to tangency points are perpendicular to the tangents, so $\triangle OPR$ and $\triangle OQR$ are right triangles. They have $OP=OQ$ (radii) and common hypotenuse $OR$, hence are congruent by RHS. Thus $PR=RQ$ and $OR$ bisects $\angle PRQ$, so $\angle PRO=60^\circ$. In right triangle $OPR$,
\[
\cos60^\circ=\frac{PR}{OR}=\frac12\quad\Longrightarrow\quad OR=2PR.
\]
Since $PR=RQ$, $\boxed{OR=PR+RQ}$.

% Q6: the inradius relation gives the semiperimeter, then the equal-tangent
% theorem reduces the two unknown sides to one unknown tangent length.
\item In Figure~\ref{fig:class10-incircle}, a triangle circumscribes a circle of radius $3\,\mathrm{cm}$. The base portions are $BD=6\,\mathrm{cm}$ and $DC=9\,\mathrm{cm}$. If area $[ABC]=54\,\mathrm{cm}^2$, find $AB$ and $AC$. \hfill (CBSE 2015, 30/1, Q6)
\begin{figure}[H]\centering
  \includegraphics[width=0.77\linewidth]{figs/IncircleTriangle}
  \caption{Circumscribed triangle with the base divided at the tangency point.}
  \label{fig:class10-incircle}
\end{figure}
\solution By the inradius relation \eqref{eq:triangle-inradius}, the semiperimeter is
\[
s=\frac{[ABC]}r=\frac{54}{3}=18\,\mathrm{cm}.
\]
Tangents from the same point are equal. If the tangent length from $A$ is $x$, then $AB=x+6$, $AC=x+9$, and $BC=6+9=15$. Hence
\[
s=\frac{(x+6)+(x+9)+15}{2}=x+15=18,
\]
so $x=3$. Therefore $\boxed{AB=9\,\mathrm{cm},\quad AC=12\,\mathrm{cm}}$.

% Q7: rewrite the polynomial as a difference of squares before solving.
\item Solve $4x^2+4bx-(a^2-b^2)=0$ for $x$. \hfill (CBSE 2015, 30/1, Q7)

\solution Rearranging and using \eqref{eq:difference-squares},
\[
(2x+b)^2-a^2=0\quad\Longrightarrow\quad (2x+b-a)(2x+b+a)=0.
\]
Thus $2x+b=\pm a$, giving
\[
\boxed{x=\frac{-b+a}{2}\quad\text{or}\quad x=\frac{-b-a}{2}}.
\]

% Q8: the two AP-sum conditions form a 2x2 linear system in a and d.
\item In an AP, $S_5+S_7=167$ and $S_{10}=235$. Find the AP. \hfill (CBSE 2015, 30/1, Q8)

\solution From \eqref{eq:ap},
\[
S_5=5a+10d,\quad S_7=7a+21d,\quad S_{10}=10a+45d.
\]
The conditions become
\[
\underbrace{\myvec{12&31\\10&45}}_{M}\myvec{a\\d}=\myvec{167\\235}.
\]
Here $\det M=12(45)-10(31)=230\ne0$, so the parameters are uniquely determined. Using the inverse matrix,
\[
\myvec{a\\d}=\frac1{230}\myvec{45&-31\\-10&12}\myvec{167\\235}=\myvec{1\\5}.
\]
The AP is $\boxed{1,6,11,16,\ldots}$.

% Q9: perpendicularity is a zero dot product.  Reject p=7 because it
% collapses B and C and does not form a triangle.
\item $A(4,7)$, $B(p,3)$, and $C(7,3)$ form a right triangle, right-angled at $B$. Find $p$. \hfill (CBSE 2015, 30/1, Q9)

\solution The side vectors are $\overrightarrow{BA}=\myvec{4-p\\4}$ and $\overrightarrow{BC}=\myvec{7-p\\0}$. Perpendicularity means their dot product is zero by \eqref{eq:vector-basics}:
\[
(4-p)(7-p)=0\quad\Longrightarrow\quad p=4\ \text{or}\ p=7.
\]
The value $p=7$ makes $B=C$, so no triangle exists. Thus $\boxed{p=4}$.

% Q10: a determinant is the coordinate-free (up to the chosen coordinates)
% test for collinearity; it also exposes the line equation immediately.
\item If $A(x,y)$, $B(-5,7)$, and $C(-4,5)$ are collinear, find the relation between $x$ and $y$. \hfill (CBSE 2015, 30/1, Q10)

\solution By the collinearity determinant \eqref{eq:collinear-det},
\[
\mydet{x+5&y-7\\1&-2}=0.
\]
Therefore $-2(x+5)-(y-7)=0$, or
\[
\boxed{2x+y+3=0}.
\]
\end{enumerate}

\subsubsection{Section C -- 3 marks each}
\begin{enumerate}[label=\textbf{Q\arabic*.},ref=\arabic*,start=11]

% Q11: turn the term relation into one equation for a,d; the sixth term
% supplies the second independent relation.
\item The $14$th term of an AP is twice its $8$th term. If the $6$th term is $-8$, find the sum of its first $20$ terms. \hfill (CBSE 2015, 30/1, Q11)

\solution Let the first term and common difference be $a,d$. By \eqref{eq:ap},
\[
a+13d=2(a+7d)\quad\Longrightarrow\quad a+d=0.
\]
Also $a+5d=-8$. Substituting $a=-d$ gives $4d=-8$, so $d=-2$ and $a=2$. Thus
\[
S_{20}=\frac{20}{2}\bigl(2(2)+19(-2)\bigr)=10(4-38)=\boxed{-340}.
\]

% Q12: retain exact radicals through the discriminant to avoid rounding.
\item Solve $\sqrt3x^2-2\sqrt2x-2\sqrt3=0$. \hfill (CBSE 2015, 30/1, Q12)

\solution Here $a=\sqrt3$, $b=-2\sqrt2$, and $c=-2\sqrt3$. Using \eqref{eq:quad-general},
\[
\Delta=(-2\sqrt2)^2-4(\sqrt3)(-2\sqrt3)=8+24=32.
\]
Hence
\[
x=\frac{2\sqrt2\pm4\sqrt2}{2\sqrt3},
\]
which gives $\boxed{x=\sqrt6\quad\text{or}\quad x=-\frac{\sqrt6}{3}}$.

% Q13: constant height converts each elevation angle to a horizontal range.
% The difference in ranges, not either full range, is the 15-second travel.
\item An aeroplane is observed at elevation $60^\circ$ from point $A$. After $15$ s, the elevation becomes $30^\circ$. At constant height $1500\sqrt3\,\mathrm m$, find its speed in km/h. \hfill (CBSE 2015, 30/1, Q13)
\begin{figure}[H]\centering
  \includegraphics[width=0.88\linewidth]{figs/Aeroplane}
  \caption{Horizontal displacement at fixed altitude.}
  \label{fig:class10-aeroplane}
\end{figure}
\solution Let the plane's initial and final horizontal ranges from $A$ be $x_1,x_2$. Since altitude $h=1500\sqrt3$ is constant, $x=h\cot\theta$:
\[
x_1=1500\sqrt3\cot60^\circ=1500\,\mathrm m,
\qquad x_2=1500\sqrt3\cot30^\circ=4500\,\mathrm m.
\]
The plane therefore travels $x_2-x_1=3000\,\mathrm m$ in $15$ s. By \eqref{eq:speed-time},
\[
v=\frac{3000}{15}=200\,\mathrm{m/s}=200\times\frac{18}{5}
=\boxed{720\,\mathrm{km/h}}.
\]

% Q14: the section formula is a vector affine combination; the 3/7 ratio
% should multiply the displacement from A towards B.
\item $A(-2,-2)$ and $B(2,-4)$. Find $P$ on segment $AB$ if $AP=\frac37 AB$. \hfill (CBSE 2015, 30/1, Q14)

\solution The vector section formula \eqref{eq:section-vector} gives
\[
\vec p=\vec a+\frac37(\vec b-\vec a)
=\myvec{-2\\-2}+\frac37\myvec{4\\-2}
=\myvec{-2/7\\-20/7}.
\]
Therefore $\boxed{P\left(-\frac27,-\frac{20}{7}\right)}$.

% Q15: probabilities give proportions of the total; the orange count
% determines the total without needing to guess the red/blue counts first.
\item A jar contains only red, blue, and orange balls. $P(R)=\frac14$, $P(B)=\frac13$, and there are $10$ orange balls. Find the total number of balls. \hfill (CBSE 2015, 30/1, Q15)

\solution Let the total be $N$. The number of red balls is $N/4$ and blue balls is $N/3$. Since these are the only colours, the orange count is
\[
N-\frac N4-\frac N3=\frac{5N}{12}=10.
\]
Thus $N=24$, so the jar contains $\boxed{24}$ balls (6 red, 8 blue, and 10 orange).

% Q16: subtract an equilateral triangle from a 60-degree sector for the
% minor segment; subtract that segment from the whole circle for the major one.
\item Find the minor and major segment areas for a circle of radius $14\,\mathrm{cm}$ and central angle $60^\circ$. Use $\pi=22/7$. \hfill (CBSE 2015, 30/1, Q16)

\solution By \eqref{eq:sector-segment}, the sector area is
\[
A_{\rm sector}=\frac{60}{360}\pi(14)^2=\frac{98\pi}{3}=\frac{308}{3}\,\mathrm{cm}^2.
\]
The triangle formed by the two radii has included angle $60^\circ$ and is equilateral with side $14$ cm; its area is $\frac{\sqrt3}{4}(14)^2=49\sqrt3\,\mathrm{cm}^2$. Hence
\[
A_{\rm minor}=\boxed{\frac{308}{3}-49\sqrt3\ \mathrm{cm}^2}.
\]
The full circle area is $\pi r^2=616\,\mathrm{cm}^2$, so
\[
A_{\rm major}=616-A_{\rm minor}
=\boxed{\frac{1540}{3}+49\sqrt3\ \mathrm{cm}^2}.
\]

% Q17: canvas covers only the cylindrical curved surface and conical curved
% surface; no floor/base is included. The cone's slant height is 3.5 m.
\item One tent has a cylindrical lower part of diameter $4.2\,\mathrm m$ and height $4\,\mathrm m$, topped by a cone of the same diameter and height $2.8\,\mathrm m$. Canvas costs Rs. 100 per m$^2$. Find the associations' 50% contribution for 100 tents, and state their values. \hfill (CBSE 2015, 30/1, Q17)
\begin{figure}[H]\centering
  \includegraphics[width=0.72\linewidth]{figs/Tent}
  \caption{Tent cross-section: a cylinder plus a cone, open at the bottom.}
  \label{fig:class10-tent}
\end{figure}
\solution The radius is $r=2.1\,\mathrm m$. The cone slant height from \eqref{eq:cone-formula} is $\ell=\sqrt{2.1^2+2.8^2}=3.5\,\mathrm m$. Canvas area per tent is the sum of the curved areas in \eqref{eq:cylinder-formula} and \eqref{eq:cone-formula}:
\[
A=2\pi rh+\pi r\ell=2\pi(2.1)(4)+\pi(2.1)(3.5)=75.9\,\mathrm{m}^2.
\]
For 100 tents, the canvas cost is $100\times75.9\times100=\text{Rs. }759{,}000$. The promised half is $\boxed{\text{Rs. }379{,}500}$. The associations show cooperation, initiative, and concern for people affected by floods.

% Q18: account explicitly for the 10% loss before dividing by 72 bottles.
\item A hemispherical bowl of internal diameter $36\,\mathrm{cm}$ is filled with liquid. The liquid is poured into 72 cylindrical bottles of diameter $6\,\mathrm{cm}$, with 10% wasted. Find each bottle's height. \hfill (CBSE 2015, 30/1, Q18)

\solution The bowl radius is $18$ cm, so its volume is $\frac23\pi18^3=3888\pi\,\mathrm{cm}^3$ by \eqref{eq:hemisphere-formula}. Only 90% remains after the transfer. Each bottle has radius $3$ cm, so its volume is $9\pi h$. Volume conservation gives
\[
72(9\pi h)=0.9(3888\pi).
\]
Cancel $\pi$ and solve: $648h=3499.2$, hence $\boxed{h=5.4\,\mathrm{cm}}$.

% Q19: subtract the top circle covered by the hemisphere; add curved hemisphere
% area. Do not count the internal circular interface as exposed surface.
\item A cube of side $10\,\mathrm{cm}$ is surmounted by a hemisphere. Find the largest possible hemisphere diameter and the cost of painting the exposed solid at Rs. 5 per $100\,\mathrm{cm}^2$; use $\pi=3.14$. \hfill (CBSE 2015, 30/1, Q19)
\begin{figure}[H]\centering
  \includegraphics[width=0.68\linewidth]{figs/CubeHemisphere}
  \caption{Hemisphere fits within the cube's top square.}
  \label{fig:class10-cube-hemi}
\end{figure}
\solution The largest circular base fitting on a $10$ cm square has diameter $10$ cm, so $r=5$ cm. The exposed area equals five complete cube faces, plus the uncovered part of the sixth face, plus the curved hemisphere area:
\[
A=5(10^2)+(10^2-\pi5^2)+2\pi5^2=600+25\pi=678.5\,\mathrm{cm}^2.
\]
At Rs. 5 per $100\,\mathrm{cm}^2$, the cost is $678.5\times5/100=\boxed{\text{Rs. }33.925\approx\text{Rs. }33.93}$.

% Q20: volume conservation simplifies after cancelling pi; the resulting
% radius is an exact cube root, 10.5 cm.
\item 504 cones of diameter $3.5\,\mathrm{cm}$ and height $3\,\mathrm{cm}$ are melted and recast into a sphere. Find its diameter and surface area; use $\pi=22/7$. \hfill (CBSE 2015, 30/1, Q20)

\solution Each cone has radius $1.75$ cm, and the total cone volume is
\[
504\left(\frac13\pi(1.75)^2(3)\right)=1543.5\pi\,\mathrm{cm}^3.
\]
Equating this to the sphere volume from \eqref{eq:sphere-formula},
\[
\frac43\pi R^3=1543.5\pi\quad\Longrightarrow\quad R^3=1157.625=(10.5)^3.
\]
Thus the diameter is $\boxed{21\,\mathrm{cm}}$. Its surface area is $4\pi(10.5)^2=441\pi=\boxed{1386\,\mathrm{cm}^2}$.
\end{enumerate}

\subsubsection{Section D -- 4 marks each}
\begin{enumerate}[label=\textbf{Q\arabic*.},ref=\arabic*,start=21]

% Q21: the diagonal is related to both sides by Pythagoras; after squaring,
% the x^2 terms partly cancel and a simple quadratic remains.
\item A rectangular field's diagonal is $16$ m longer than its shorter side. Its longer side is $14$ m longer than its shorter side. Find the side lengths. \hfill (CBSE 2015, 30/1, Q21)

\solution Let the shorter side be $x>0$. The longer side is $x+14$ and diagonal $x+16$. Pythagoras gives
\[
x^2+(x+14)^2=(x+16)^2\quad\Longrightarrow\quad x^2-4x-60=0.
\]
Factor: $(x-10)(x+6)=0$. The positive length is $x=10$ m, and the longer side is $24$ m. Therefore $\boxed{10\,\mathrm m\text{ by }24\,\mathrm m}$ (diagonal $26$ m).

% Q22: the last ten terms are T_51,...,T_60; use the end-term sum form
% of the AP formula rather than separately adding ten numbers.
\item Find the 60th term of the AP $8,10,12,\ldots$ and the sum of its last 10 terms. \hfill (CBSE 2015, 30/1, Q22)

\solution Here $a=8$ and $d=2$. By \eqref{eq:ap},
\[
t_{60}=8+59(2)=\boxed{126}.
\]
The 51st term is $t_{51}=8+50(2)=108$. Therefore the last ten terms form an AP with first term 108 and last term 126; their sum is
\[
S_{\rm last\ 10}=\frac{10}{2}(108+126)=\boxed{1170}.
\]

% Q23: distances divided by speeds produce the total-time equation.
% The negative root violates the meaning of speed and is discarded.
\item A train travels 54 km at speed $x$ km/h and then 63 km at $(x+6)$ km/h. Total journey time is 3 h. Find $x$. \hfill (CBSE 2015, 30/1, Q23)

\solution Using time $=\text{distance}/\text{speed}$ from \eqref{eq:speed-time},
\[
\frac{54}{x}+\frac{63}{x+6}=3.
\]
Multiplying through by $x(x+6)$ gives $54(x+6)+63x=3x(x+6)$, hence
\[
x^2-33x-108=0=(x-36)(x+3).
\]
Only a positive speed is admissible, so $\boxed{x=36\,\mathrm{km/h}}$.

% Q24: RHS congruence is the standard proof and makes the equal-tangent
% result from the formula sheet explicit rather than merely citing it.
\item Prove that tangent lengths drawn from an external point to a circle are equal. \hfill (CBSE 2015, 30/1, Q24)

\solution Let $PA$ and $PB$ touch the circle at $A$ and $B$, with centre $O$. Join $OA$, $OB$, and $OP$. Since a radius is perpendicular to a tangent, $\angle OAP=\angle OBP=90^\circ$. Also $OA=OB$ (radii) and $OP$ is common. Hence $\triangle OAP\cong\triangle OBP$ by RHS, so corresponding sides satisfy
\[
\boxed{PA=PB}.
\]

% Q25: the radius through the midpoint of an arc bisects its central angle
% and is perpendicular to the corresponding chord; the tangent is also
% perpendicular to that radius, so the two lines are parallel.
\item Prove that the tangent at the midpoint of an arc is parallel to the chord joining the arc's endpoints. \hfill (CBSE 2015, 30/1, Q25)

\solution Let $M$ be the midpoint of arc $AB$ and $O$ be the centre. Because arcs $AM$ and $MB$ are equal, their chords are equal: $MA=MB$. In isosceles triangle $AMB$, the line $OM$ through the arc midpoint is the perpendicular bisector of $AB$, so $OM\perp AB$. The tangent at $M$ is perpendicular to radius $OM$ as well. Two lines perpendicular to the same line are parallel; therefore
\[
\boxed{\text{tangent at }M\parallel AB}.
\]

% Q26: give a reproducible straightedge-and-compass construction.  Eight
% equal marks on an auxiliary ray create the scale factor 8/6=4/3 by parallels.
\item Construct $\triangle ABC$ with $AB=6$ cm, $\angle A=30^\circ$, $\angle B=60^\circ$. Then construct a similar triangle $AB'C'$ with $AB'=8$ cm. \hfill (CBSE 2015, 30/1, Q26)

\solution
\begin{enumerate}[label=(\alph*)]
\item Draw $AB=6$ cm. At $A$, construct a $30^\circ$ ray to $AB$; at $B$, construct a $60^\circ$ ray to $BA$. Their intersection is $C$.
\item From $A$, draw an auxiliary ray $AX$ not collinear with $AB$. Mark eight equal intervals on it to obtain $A_1,\ldots,A_8$.
\item Join $A_6$ to $B$. Through $A_8$, draw a line parallel to $A_6B$; let it meet the ray $AB$ produced at $B'$. By the intercept theorem, $AB'/AB=AA_8/AA_6=8/6$, hence $AB'=8$ cm.
\item Through $B'$, draw a line parallel to $BC$ to meet the ray $AC$ produced at $C'$. Then $\triangle AB'C'\sim\triangle ABC$ by corresponding angles, and the required construction is complete.
\end{enumerate}

% Q27: use two tangent equations for cloud height and horizontal distance.
% The reflection is as far below the water as the real cloud is above it.
\item Point $A$ is 20 m above a lake. The elevation angle of a cloud is $30^\circ$, and the depression angle of its reflection is $60^\circ$. Find the distance from $A$ to the cloud. \hfill (CBSE 2015, 30/1, Q27)
\begin{figure}[H]\centering
  \includegraphics[width=0.86\linewidth]{figs/CloudReflection}
  \caption{Cloud and its reflection are symmetric about the lake surface.}
  \label{fig:class10-cloud}
\end{figure}
\solution Let the cloud height above water be $H$ and the horizontal distance be $x$. The two angle conditions give
\[
\frac{H-20}{x}=\tan30^\circ=\frac1{\sqrt3},\qquad
\frac{H+20}{x}=\tan60^\circ=\sqrt3.
\]
Equivalently, a linear system in $(H,x)$ is
\[
\myvec{1&-1/\sqrt3\\1&-\sqrt3}\myvec{H\\x}=\myvec{20\\-20}.
\]
Solving gives $H=40$ m and $x=20\sqrt3$ m. Since the cloud is $H-20=20$ m above $A$ vertically,
\[
AC=\sqrt{x^2+(H-20)^2}=\sqrt{(20\sqrt3)^2+20^2}=\boxed{40\,\mathrm m}.
\]

% Q28: use inclusion--exclusion for spades or aces; use the complement
% for neither jack nor king; kings and queens are disjoint categories.
\item A card is drawn from a well-shuffled 52-card deck. Find the probability that it is (i) a spade or ace, (ii) a black king, (iii) neither a jack nor a king, (iv) either a king or a queen. \hfill (CBSE 2015, 30/1, Q28)

\solution By \eqref{eq:prob-classical} and \eqref{eq:prob-union}:
\begin{enumerate}[label=(\roman*)]
\item $13$ spades and $4$ aces overlap at the ace of spades, so $P=(13+4-1)/52=\boxed{4/13}$.
\item There are two black kings, so $P=2/52=\boxed{1/26}$.
\item Four jacks and four kings are disjoint, so $P=(52-4-4)/52=\boxed{11/13}$.
\item There are four kings and four queens, so $P=(4+4)/52=\boxed{2/13}$.
\end{enumerate}

% Q29: the determinant computes twice the area.  Solve the signed area
% equation and retain both real parameter values.
\item Find $k$ if the triangle with vertices $(1,-1)$, $(-4,2k)$, and $(-k,-5)$ has area $24$ square units. \hfill (CBSE 2015, 30/1, Q29)

\solution By the determinant area formula \eqref{eq:triangle-det},
\[
2[ABC]=\left|\mydet{-5&2k+1\\-k-1&-4}\right|
=\left|2k^2+3k+21\right|.
\]
Since the quadratic has positive leading coefficient and negative discriminant, it is positive for every real $k$. Thus $2k^2+3k+21=48$, or
\[
2k^2+3k-27=0=(2k+9)(k-3)
\]
Solving directly gives discriminant $225$, hence $k=(-3\pm15)/4$. Therefore
\[
\boxed{k=3\quad\text{or}\quad k=-\frac92}.
\]

% Q30: the circle through each vertical side has centre O and radius half the
% square diagonal. Each shaded piece is a 90-degree circular segment.
\item A square lawn has side $42$ m. Two circular flower beds lie outside the vertical sides, each circle centred at the square's diagonal intersection $O$. Find the total shaded area. \hfill (CBSE 2015, 30/1, Q30)
\begin{figure}[H]\centering
  \includegraphics[width=0.79\linewidth]{figs/FlowerBeds}
  \caption{Two congruent circular segments outside the square.}
  \label{fig:class10-flowerbeds}
\end{figure}
\solution The centre-to-corner radius is half the diagonal, so
\[
r^2=\frac{42^2+42^2}{4}=882\,\mathrm{m}^2.
\]
Each side subtends $90^\circ$ at the centre. By \eqref{eq:sector-segment}, one segment's area is $(90/360)\pi r^2-\frac12r^2=\frac14\pi(882)-441$. There are two equal beds, hence
\[
A_{\rm total}=2\left(\frac{882\pi}{4}-441\right)=441\pi-882.
\]
With $\pi=22/7$, $A_{\rm total}=1386-882=\boxed{504\,\mathrm{m}^2}$.

% Q31: remove the volume of two hemispheres (one sphere), then divide by the
% cross-sectional area of a wire whose 1.4 cm thickness is its diameter.
\item From both ends of a metal cylinder, hemispherical cavities of the same diameter are scooped out. The cylinder has height $10$ cm and radius $4.2$ cm. The remaining metal is made into a cylindrical wire of thickness $1.4$ cm. Find the wire length; use $\pi=22/7$. \hfill (CBSE 2015, 30/1, Q31)
\begin{figure}[H]\centering
  \includegraphics[width=0.83\linewidth]{figs/CylinderScooped}
  \caption{The removed pair of hemispheres has the volume of one sphere.}
  \label{fig:class10-scooped-cylinder}
\end{figure}
\solution The two hemispherical cavities together remove the volume of one sphere. From \eqref{eq:scooped-cylinder}, the remaining volume is
\[
V=\pi(4.2)^2(10)-\frac43\pi(4.2)^3=77.616\pi\,\mathrm{cm}^3.
\]
A wire thickness of $1.4$ cm means radius $0.7$ cm. Equating the remaining volume to the wire volume $\pi(0.7)^2L$ gives
\[
L=\frac{77.616\pi}{0.49\pi}=\boxed{158.4\,\mathrm{cm}}.
\]
\end{enumerate}
